# Runtime dependencies (pip-installable straight on aarch64 / JetPack 6~7)
#
# Why TensorRT is not lumped into this single requirements file:
# tensorrt / cuda-python must match JetPack's own CUDA version exactly; pip-installing a mismatched
# version falls back to CPU silently (OCR goes from 300ms to 3s with no error). Use the JetPack
# builds, and install separately per docs/ocr-gpu.md only if you really want GPU-accelerated OCR.

numpy>=1.26,<3
scipy>=1.11

# ASR / VAD / wake word: one sherpa-onnx stack covers all three (SenseVoice + Silero + KWS zipformer)
sherpa-onnx==2.0.1

# OCR ("glance at the screen")
opencv-python-headless>=4.9
rapidocr-onnxruntime>=1.3

# TTS
#   piper-tts 1.8.0+ no longer depends on piper-phonemize and has an aarch64 wheel, install directly
#   Japanese voices need OpenJTalk tokenization: upstream pyopenjtalk is source-only, so use pyopenjtalk-plus
#   (module name is still pyopenjtalk, drop-in)
piper-tts>=1.8.0
pyopenjtalk-plus>=0.4.1 ; platform_machine == "aarch64"
pyopenjtalk>=0.3 ; platform_machine != "aarch64"

# Web
fastapi>=0.110
uvicorn>=0.27
python-multipart>=0.0.9

# Tests
pytest>=7.4
